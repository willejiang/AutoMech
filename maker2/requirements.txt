# maker2 host-side Python deps (the boss -> managers -> workers -> assembler
# pipeline). The geometry worker defaults to CadQuery (settings.worker_backend);
# OpenSCAD is a legacy fallback and needs the OpenSCAD CLI (OPENSCAD_BIN) instead.
# FreeCAD modules (FreeCAD/Part/Sketcher/...) are provided by a FreeCAD install,
# not pip. Evaluator/physics deps live in evaluator/requirements.txt.
cadquery         # parametric CAD kernel (OpenCASCADE): the default geometry worker
numpy
py-slvs==1.0.6  # authoritative cross-sub constraint solver; GPL-3.0
openai
trimesh          # STL validation (validation.check_stl) + mesh I/O + convex decomp
manifold3d       # exact mesh boolean engine for the rigid-conflict gate (subcheck):
                 # real SOLID-overlap so hollow bores/keyways aren't false-flagged. If
                 # absent, precheck._solid_intersection_frac degrades to an AABB proxy.
yourdfpy         # URDF load/validate
six
# MuJoCo pure-contact path (settings.engine="mujoco"). `mujoco` is required for that
# path; `coacd` is an OPTIONAL quality upgrade for convex decomposition — without it,
# convex_decomp falls back to trimesh V-HACD, then a convex hull (which sets
# metrics["contact_degraded"]). The default engine is "pybullet" and needs neither.
mujoco           # pure-contact physics (only imported when engine="mujoco")
# coacd          # optional: better convex decomposition than the V-HACD/hull fallback
# Local offline RAG (settings.enable_kb; maker2/kb). CPU-only, no network at runtime.
# Only imported when enable_kb is on; the pipeline runs without them otherwise.
sentence-transformers   # all-MiniLM-L6-v2 embedder (~80MB), CPU
faiss-cpu               # flat inner-product vector index per collection
